# step1
- listの中の値ごとにカウントしていく
- カウントしたものをソートして、カウントが大きいもの順に並べる
- 上位k個を取り出す

```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        counts = {}

        for num in nums:
            counts[num] = 1 + counts.get(num, 0)
        
        ordered = sorted(counts, key=counts.get, reverse=True)
        return ordered[:k]
        
```

```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        # リストの各要素を数える.例えば[1,1,1,2,2,3]のときのcounterはCounter({1: 3, 2: 2, 3: 1})
        counter = Counter(nums)
        print(counter)
        # most_common(k)でk番目まで抽出し、pairに入れるので
        # [(1, 3),(2, 2)]が１個ずつpairに入り、pair[0]で1,2を取ってリストに入れてreturn
        return [pair[0] for pair in counter.most_common(k)]
```
