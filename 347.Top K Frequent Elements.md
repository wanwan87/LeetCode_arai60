# step1
- listの中の値ごとにカウントしていく
- カウントしたものをソートして、カウントが大きいもの順に並べる
- 上位k個を取り出す

```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        counts = {}

        for num in nums:
            counts[num] = 1 + counts.get(num, 0)
        
        ordered = sorted(counts, key=counts.get, reverse=True)
        return ordered[:k]
        
```

counterの方法

```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        # リストの各要素を数える.例えば[1,1,1,2,2,3]のときのcounterはCounter({1: 3, 2: 2, 3: 1})
        counter = Counter(nums)
        print(counter)
        # most_common(k)でk番目まで抽出し、pairに入れるので
        # [(1, 3),(2, 2)]が１個ずつpairに入り、pair[0]で1,2を取ってリストに入れてreturn
        return [pair[0] for pair in counter.most_common(k)]
```

Quick select
```python
    def quick_select(arr, k):
        # 配列をどこで分割するか
        pivot = arr[0]
        right = []
        left = []
        for x in arr[1:]:
            # pivotがxより小さい場合、左の配列に入れる.
            if x < pivot:
                left.append(x)
            else:
                right.append(x)
        # leftの数がk番目よりも多い場合、leftの中に答えがあるため探索範囲をleftで再探索
        if k <= len(left):
            return quick_select(left, k)
        # leftの次、つまりpivotがちょうどk番目だった場合
        elif k == len(left) + 1:
            return pivot
        # 答えはrightの中なので、rightの中を、k-捨てた分のleft-pivot1個捨てた。
        else:
            return quick_select(right, k-len(left)-1)

```

```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        element_counter = {}
        for num in nums:
            element_counter[num] = 1 + element_counter.get(num, 0)

        heap = []
        for num, frequency in element_counter.items():
            heapq.heappush(heap, (frequency, num))
            # heapで小さい順に並べて、kよりも多い場合はpopで捨てていく
            if len(heap) > k:
                heapq.heappop(heap)

        res = []
        # 使わない変数はアンダースコアで受ける
        for _, num in heap:
            res.append(num)
        
```
